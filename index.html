import os from functools import wraps from flask import Flask, render_template, redirect, url_for, flash, session, request from flask_wtf.csrf import CSRFProtect import psycopg2 from psycopg2.extras import RealDictCursor from dotenv import load_dotenv # Importación de formularios personalizados from forms.login_form import LoginForm from forms.categoria_form import CategoriaForm from forms.producto_form import ProductoForm # Cargar variables de entorno load_dotenv() # Configuración de Flask app = Flask(__name__) app.config["SECRET_KEY"] = os.environ.get("SECRET_KEY", "clave-secreta-desarrollo") app.config["DATABASE_URL"] = os.environ.get("DATABASE_URL") # Protección CSRF csrf = CSRFProtect(app) # Conexión con PostgreSQL def get_connection(): database_url = app.config["DATABASE_URL"] if not database_url: raise RuntimeError("No se encontró DATABASE_URL.") return psycopg2.connect(database_url) # Decorador de Login Requerido def login_required(view): @wraps(view) def wrapped_view(*args, **kwargs): if "usuario_id" not in session: flash("Debe iniciar sesión para acceder.", "warning") return redirect(url_for("login")) return view(*args, **kwargs) return wrapped_view # Página Principal @app.route("/") def index(): return render_template("index.html") # Login @app.route("/login", methods=["GET", "POST"]) def login(): if "usuario_id" in session: return redirect(url_for("productos")) form = LoginForm() if form.validate_on_submit(): correo = form.correo.data.strip().lower() password = form.password.data connection = None try: connection = get_connection() with connection.cursor(cursor_factory=RealDictCursor) as cursor: cursor.execute("SELECT id, nombre, correo, password FROM usuarios WHERE correo = %s", (correo,)) usuario = cursor.fetchone() if usuario and usuario["password"] == password: session.clear() session["usuario_id"] = usuario["id"] session["usuario_nombre"] = usuario["nombre"] session["usuario_correo"] = usuario["correo"] flash("Inicio de sesión correcto.", "success") return redirect(url_for("productos")) else: flash("Correo o contraseña incorrectos.", "danger") except Exception as error: print("Error de login:", error) flash("Ocurrió un error al iniciar sesión.", "danger") finally: if connection: connection.close() return render_template("login.html", form=form) # Cerrar Sesión @app.route("/logout") def logout(): session.clear() flash("Sesión cerrada correctamente.", "info") return redirect(url_for("login")) # Listar Categorías @app.route("/categorias") @login_required def categorias(): connection = None try: connection = get_connection() with connection.cursor(cursor_factory=RealDictCursor) as cursor: cursor.execute("SELECT id, nombre, descripcion FROM categorias ORDER BY id DESC") categorias = cursor.fetchall() return render_template("categorias/lista.html", categorias=categorias) except Exception as error: print("Error:", error) flash("No se pudieron cargar las categorías.", "danger") return redirect(url_for("productos")) finally: if connection: connection.close() # Crear Categoría @app.route("/categorias/nueva", methods=["GET", "POST"]) @login_required def nueva_categoria(): form = CategoriaForm() if form.validate_on_submit(): nombre = form.nombre.data.strip() descripcion = form.descripcion.data.strip() if form.descripcion.data else "" connection = None try: connection = get_connection() with connection.cursor() as cursor: cursor.execute( "INSERT INTO categorias (nombre, descripcion) VALUES (%s, %s)", (nombre, descripcion) ) connection.commit() flash("Categoría creada correctamente.", "success") return redirect(url_for("categorias")) except psycopg2.errors.UniqueViolation: if connection: connection.rollback() flash("La categoría ya existe.", "warning") except Exception as error: if connection: connection.rollback() print("Error:", error) flash("No se pudo crear la categoría.", "danger") finally: if connection: connection.close() return render_template("categorias/formulario.html", form=form, titulo="Nueva Categoría") if __name__ == "__main__": app.run(debug=True)
