import os import sqlite3 from flask import Flask, flash, redirect, render_template, request, url_for from flask_wtf import FlaskForm from wtforms import FloatField, IntegerField, StringField from wtforms.validators import DataRequired, NumberRange # Inicialización de la aplicación Flask app = Flask(__name__) app.config['SECRET_KEY'] = 'clave_secreta_proyecto_integrador_2026' # Configuración de la ruta de la base de datos dentro de la carpeta 'data' DATABASE = os.path.join('data', 'ferreteria.db') def get_db_connection(): """Establece la conexión con la base de datos SQLite y configura row_factory para acceder a las columnas por nombre. """ conn = sqlite3.connect(DATABASE) conn.row_factory = sqlite3.Row return conn def init_db(): """Crea la carpeta 'data' si no existe y la tabla 'productos'""" os.makedirs('data', exist_ok=True) conn = get_db_connection() conn.execute(''' CREATE TABLE IF NOT EXISTS productos ( id INTEGER PRIMARY KEY AUTOINCREMENT, nombre TEXT NOT NULL, precio REAL NOT NULL, stock INTEGER NOT NULL ) ''') conn.commit() conn.close() # ========================================== # DEFINICIÓN DE FORMULARIOS (Flask-WTF) # ========================================== class ProductoForm(FlaskForm): nombre = StringField('Nombre del Producto', validators=[DataRequired()]) precio = FloatField( 'Precio', validators=[DataRequired(), NumberRange(min=0.01)] ) stock = IntegerField('Stock', validators=[DataRequired(), NumberRange(min=0)]) # ========================================== # RUTAS DE LA APLICACIÓN # ========================================== @app.route('/') def index(): return render_template('index.html') # MÓDULO DE PRODUCTOS (Flujo completo: INSERT -> SELECT -> Tabla HTML) @app.route('/productos') def listar_productos(): # SELECT: Recuperar todos los productos registrados de la base de datos conn = get_db_connection() productos = conn.execute('SELECT * FROM productos').fetchall() conn.close() return render_template('productos.html', productos=productos) @app.route('/productos/nuevo', methods=['GET', 'POST']) def nuevo_producto(): form = ProductoForm() # Validación del lado del servidor (Flask-WTF + Método POST) if form.validate_on_submit(): nombre = form.nombre.data precio = form.precio.data stock = form.stock.data # INSERT: Almacenar de forma persistente en SQLite conn = get_db_connection() conn.execute( 'INSERT INTO productos (nombre, precio, stock) VALUES (?, ?, ?)', (nombre, precio, stock), ) conn.commit() conn.close() flash('¡Producto registrado y guardado exitosamente en SQLite!', 'success') return redirect(url_for('listar_productos')) return render_template('formulario_producto.html', form=form) # Rutas preparadas para los demás módulos (Clientes, Proveedores, Facturación) @app.route('/clientes') def listar_clientes(): return render_template('clientes.html') @app.route('/proveedores') def listar_proveedores(): return render_template('proveedores.html') @app.route('/facturacion') def facturacion(): return render_template('facturacion.html') # ========================================== # EJECUCIÓN DEL SERVIDOR # ========================================== if __name__ == '__main__': # Inicializar la base de datos antes de arrancar el servidor init_db() app.run(debug=True)
