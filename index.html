import os from functools import wraps from flask import ( Flask, render_template, redirect, url_for, flash, session, request ) from flask_wtf.csrf import CSRFProtect import psycopg2 from psycopg2.extras import RealDictCursor from dotenv import load_dotenv from forms.login_form import LoginForm from forms.categoria_form import CategoriaForm from forms.producto_form import ProductoForm # ========================================================== # CARGAR VARIABLES DE ENTORNO # ========================================================== load_dotenv() # ========================================================== # CONFIGURACIÓN DE FLASK # ========================================================== app = Flask(__name__) app.config["SECRET_KEY"] = os.environ.get( "SECRET_KEY", "clave-secreta-desarrollo" ) app.config["DATABASE_URL"] = os.environ.get( "DATABASE_URL" ) # ========================================================== # PROTECCIÓN CSRF # ========================================================== csrf = CSRFProtect(app) # ========================================================== # CONEXIÓN CON POSTGRESQL # ========================================================== def get_connection(): database_url = app.config["DATABASE_URL"] if not database_url: raise RuntimeError( "No se encontró DATABASE_URL." ) return psycopg2.connect(database_url) # ========================================================== # DECORADOR LOGIN REQUIRED # ========================================================== def login_required(view): @wraps(view) def wrapped_view(*args, **kwargs): if "usuario_id" not in session: flash( "Debe iniciar sesión para acceder.", "warning" ) return redirect( url_for("login") ) return view(*args, **kwargs) return wrapped_view # ========================================================== # PÁGINA PRINCIPAL # ========================================================== @app.route("/") def index(): return render_template( "index.html" ) # ========================================================== # LOGIN # ========================================================== @app.route( "/login", methods=["GET", "POST"] ) def login(): if "usuario_id" in session: return redirect( url_for("productos") ) form = LoginForm() if form.validate_on_submit(): correo = form.correo.data.strip().lower() password = form.password.data connection = None try: connection = get_connection() with connection.cursor( cursor_factory=RealDictCursor ) as cursor: cursor.execute( """ SELECT id, nombre, correo, password FROM usuarios WHERE correo = %s """, (correo,) ) usuario = cursor.fetchone() if ( usuario and usuario["password"] == password ): session.clear() session["usuario_id"] = usuario["id"] session["usuario_nombre"] = usuario["nombre"] session["usuario_correo"] = usuario["correo"] flash( "Inicio de sesión correcto.", "success" ) return redirect( url_for("productos") ) flash( "Correo o contraseña incorrectos.", "danger" ) except Exception as error: print( "Error de login:", error ) flash( "Ocurrió un error al iniciar sesión.", "danger" ) finally: if connection: connection.close() return render_template( "login.html", form=form ) # ========================================================== # CERRAR SESIÓN # ========================================================== @app.route("/logout") def logout(): session.clear() flash( "Sesión cerrada correctamente.", "info" ) return redirect( url_for("login") ) # ========================================================== # ===================== CATEGORÍAS ========================= # ========================================================== # ========================================================== # LISTAR CATEGORÍAS # ========================================================== @app.route("/categorias") @login_required def categorias(): connection = None try: connection = get_connection() with connection.cursor( cursor_factory=RealDictCursor ) as cursor: cursor.execute( """ SELECT id, nombre, descripcion FROM categorias ORDER BY id DESC """ ) categorias = cursor.fetchall() return render_template( "categorias/lista.html", categorias=categorias ) except Exception as error: print( "Error:", error ) flash( "No se pudieron cargar las categorías.", "danger" ) return redirect( url_for("productos") ) finally: if connection: connection.close() # ========================================================== # CREAR CATEGORÍA # ========================================================== @app.route( "/categorias/nueva", methods=["GET", "POST"] ) @login_required def nueva_categoria(): form = CategoriaForm() if form.validate_on_submit(): nombre = form.nombre.data.strip() descripcion = ( form.descripcion.data.strip() if form.descripcion.data else "" ) connection = None try: connection = get_connection() with connection.cursor() as cursor: cursor.execute( """ INSERT INTO categorias ( nombre, descripcion ) VALUES ( %s, %s ) """, ( nombre, descripcion ) ) connection.commit() flash( "Categoría creada correctamente.", "success" ) return redirect( url_for("categorias") ) except psycopg2.errors.UniqueViolation: if connection: connection.rollback() flash( "La categoría ya existe.", "warning" ) except Exception as error: if connection: connection.rollback() print( "Error:", error ) flash( "No se pudo crear la categoría.", "danger" ) finally: if connection: connection.close() return render_template( "categorias/formulario.html", form=form, titulo="Nueva categoría" ) # ========================================================== # EDITAR CATEGORÍA # ========================================================== @app.route( "/categorias/editar/", methods=["GET", "POST"] ) @login_required def editar_categoria(id): connection = None try: connection = get_connection() with connection.cursor( cursor_factory=RealDictCursor ) as cursor: cursor.execute( """ SELECT id, nombre, descripcion FROM categorias WHERE id = %s """, (id,) ) categoria = cursor.fetchone() if not categoria: flash( "Categoría no encontrada.", "warning" ) return redirect( url_for("categorias") ) form = CategoriaForm() if request.method == "GET": form.nombre.data = categoria["nombre"] form.descripcion.data = ( categoria["descripcion"] or "" ) if form.validate_on_submit(): nombre = form.nombre.data.strip() descripcion = ( form.descripcion.data.strip() if form.descripcion.data else "" ) with connection.cursor() as cursor: cursor.execute( """ UPDATE categorias SET nombre = %s, descripcion = %s WHERE id = %s """, ( nombre, descripcion, id ) ) connection.commit() flash( "Categoría actualizada correctamente.", "success" ) return redirect( url_for("categorias") ) return render_template( "categorias/formulario.html", form=form, titulo="Editar categoría" ) except psycopg2.errors.UniqueViolation: if connection: connection.rollback() flash( "Ya existe una categoría con ese nombre.", "warning" ) return redirect( url_for( "editar_categoria", id=id ) ) except Exception as error: if connection: connection.rollback() print( "Error:", error ) flash( "No se pudo actualizar la categoría.", "danger" ) return redirect( url_for("categorias") ) finally: if connection: connection.close() # ========================================================== # ELIMINAR CATEGORÍA # ========================================================== @app.route( "/categorias/eliminar/", methods=["POST"] ) @login_required def eliminar_categoria(id): connection = None try: connection = get_connection() with connection.cursor() as cursor: cursor.execute( """ DELETE FROM categorias WHERE id = %s """, (id,) ) connection.commit() flash( "Categoría eliminada correctamente.", "success" ) except psycopg2.errors.ForeignKeyViolation: if connection: connection.rollback() flash( "No puede eliminar esta categoría porque tiene productos relacionados.", "warning" ) except Exception as error: if connection: connection.rollback() print( "Error:", error ) flash( "No se pudo eliminar la categoría.", "danger" ) finally: if connection: connection.close() return redirect( url_for("categorias") ) # ========================================================== # ====================== PRODUCTOS ========================== # ========================================================== # ========================================================== # OBTENER CATEGORÍAS PARA EL FORMULARIO # ========================================================== def obtener_categorias(): connection = None try: connection = get_connection() with connection.cursor( cursor_factory=RealDictCursor ) as cursor: cursor.execute( """ SELECT id, nombre FROM categorias ORDER BY nombre """ ) categorias = cursor.fetchall() return [ ( categoria["id"], categoria["nombre"] ) for categoria in categorias ] finally: if connection: connection.close() # ========================================================== # LISTAR PRODUCTOS # AQUÍ SE UTILIZA JOIN # ========================================================== @app.route("/productos") @login_required def productos(): connection = None try: connection = get_connection() with connection.cursor( cursor_factory=RealDictCursor ) as cursor: cursor.execute( """ SELECT p.id, p.nombre, p.descripcion, p.precio, p.categoria_id, c.nombre AS categoria FROM productos p INNER JOIN categorias c ON p.categoria_id = c.id ORDER BY p.id DESC """ ) productos = cursor.fetchall() return render_template( "productos/lista.html", productos=productos ) except Exception as error: print( "Error:", error ) flash( "No se pudieron cargar los productos.", "danger" ) return render_template( "productos/lista.html", productos=[] ) finally: if connection: connection.close() # ========================================================== # CREAR PRODUCTO # ========================================================== @app.route( "/productos/nuevo", methods=["GET", "POST"] ) @login_required def nuevo_producto(): form = ProductoForm() try: form.categoria_id.choices = obtener_categorias() except Exception as error: print( "Error:", error ) flash( "No se pudieron cargar las categorías.", "danger" ) return redirect( url_for("productos") ) if form.validate_on_submit(): nombre = form.nombre.data.strip() descripcion = ( form.descripcion.data.strip() if form.descripcion.data else "" ) precio = form.precio.data categoria_id = form.categoria_id.data connection = None try: connection = get_connection() with connection.cursor() as cursor: cursor.execute( """ INSERT INTO productos ( nombre, descripcion, precio, categoria_id ) VALUES ( %s, %s, %s, %s ) """, ( nombre, descripcion, precio, categoria_id ) ) connection.commit() flash( "Producto creado correctamente.", "success" ) return redirect( url_for("productos") ) except Exception as error: if connection: connection.rollback() print( "Error:", error ) flash( "No se pudo crear el producto.", "danger" ) finally: if connection: connection.close() return render_template( "productos/formulario.html", form=form, titulo="Nuevo producto" ) # ========================================================== # EDITAR PRODUCTO # ========================================================== @app.route( "/productos/editar/", methods=["GET", "POST"] ) @login_required def editar_producto(id): connection = None try: connection = get_connection() with connection.cursor( cursor_factory=RealDictCursor ) as cursor: cursor.execute( """ SELECT id, nombre, descripcion, precio, categoria_id FROM productos WHERE id = %s """, (id,) ) producto = cursor.fetchone() if not producto: flash( "Producto no encontrado.", "warning" ) return redirect( url_for("productos") ) form = ProductoForm() form.categoria_id.choices = obtener_categorias() if request.method == "GET": form.nombre.data = producto["nombre"] form.descripcion.data = ( producto["descripcion"] or "" ) form.precio.data = producto["precio"] form.categoria_id.data = ( producto["categoria_id"] ) if form.validate_on_submit(): nombre = form.nombre.data.strip() descripcion = ( form.descripcion.data.strip() if form.descripcion.data else "" ) precio = form.precio.data categoria_id = form.categoria_id.data with connection.cursor() as cursor: cursor.execute( """ UPDATE productos SET nombre = %s, descripcion = %s, precio = %s, categoria_id = %s WHERE id = %s """, ( nombre, descripcion, precio, categoria_id, id ) ) connection.commit() flash( "Producto actualizado correctamente.", "success" ) return redirect( url_for("productos") ) return render_template( "productos/formulario.html", form=form, titulo="Editar producto" ) except Exception as error: if connection: connection.rollback() print( "Error:", error ) flash( "No se pudo actualizar el producto.", "danger" ) return redirect( url_for("productos") ) finally: if connection: connection.close() # ========================================================== # ELIMINAR PRODUCTO # ========================================================== @app.route( "/productos/eliminar/", methods=["POST"] ) @login_required def eliminar_producto(id): connection = None try: connection = get_connection() with connection.cursor() as cursor: cursor.execute( """ DELETE FROM productos WHERE id = %s """, (id,) ) connection.commit() flash( "Producto eliminado correctamente.", "success" ) except Exception as error: if connection: connection.rollback() print( "Error:", error ) flash( "No se pudo eliminar el producto.", "danger" ) finally: if connection: connection.close() return redirect( url_for("productos") ) # ========================================================== # EJECUTAR APLICACIÓN # ========================================================== if __name__ == "__main__": app.run( debug=True, host="0.0.0.0", port=int( os.environ.get( "PORT", 5000 ) ) )
